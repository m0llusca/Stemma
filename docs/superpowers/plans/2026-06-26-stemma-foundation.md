# Stemma Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Stemma foundation layer: login-first entry, workspace localization with online editing, fast shell/data-loading boundaries for all authenticated pages, semantic status colors, integration install contracts, and runtime regression guards.

**Architecture:** Add platform primitives first, then migrate pages onto them. The localization runtime is server-first with built-in dictionary fallback and database overrides; the shell uses a lightweight snapshot plus async page data boundaries; semantic color decisions live in typed helpers and shared components. Integration install contracts describe readiness honestly without implementing every OAuth flow in this phase.

**Tech Stack:** Next.js 16 App Router, React 19, Prisma 6/PostgreSQL, Vitest, Playwright, TypeScript, Tailwind/CSS modules already present in `apps/web`.

```ts
const localeCodePattern = /^[a-z]{2,3}(?:-[A-Z][a-z]{3})?(?:-[A-Z]{2}|-[0-9]{3})?$/;

export function normalizeLocaleCode(value: string) {
  const trimmed = value.trim();
  const parts = trimmed.split("-");
  const normalized = parts
    .map((part, index) => {
      if (index === 0) {
        return part.toLowerCase();
      }
      if (part.length === 2) {
        return part.toUpperCase();
      }
      if (part.length === 4) {
        return `${part.slice(0, 1).toUpperCase()}${part.slice(1).toLowerCase()}`;
      }
      return part;
    })
    .join("-");

  if (!localeCodePattern.test(normalized)) {
    throw new Error("Некорректный код языка.");
  }

  return normalized;
}
```

- [ ] **Step 6: Create dictionary loader**

Create `apps/web/src/lib/i18n/dictionary.ts`:

```ts
import { prisma } from "@/lib/db";
import { builtInDefaultLocale, builtInDictionaries } from "@/lib/i18n/built-in";
import { normalizeLocaleCode } from "@/lib/i18n/locale-codes";

export type Dictionary = {
  locale: string;
  t(key: string): string;
  entries: Record<string, string>;
};

function builtInEntries(locale: string) {
  return {
    ...builtInDictionaries[builtInDefaultLocale],
    ...(locale in builtInDictionaries ? builtInDictionaries[locale as keyof typeof builtInDictionaries] : {})
  };
}

export async function getDictionary(workspaceId: string, requestedLocale: string): Promise<Dictionary> {
  const localeCode = normalizeLocaleCode(requestedLocale);
  const requested = await prisma.locale.findFirst({
    where: { workspaceId, code: localeCode, isEnabled: true },
    select: { id: true, code: true }
  });
  const defaults = await prisma.locale.findMany({
    where: { workspaceId, isDefault: true, isEnabled: true },
    select: { id: true, code: true },
    take: 1
  });
  const activeLocale = requested ?? defaults[0] ?? { id: "", code: builtInDefaultLocale };
  const values = activeLocale.id
    ? await prisma.translationValue.findMany({
        where: {
          workspaceId,
          localeId: activeLocale.id,
          publishedText: { not: null }
        },
        select: {
          publishedText: true,
          key: { select: { namespace: true, key: true, defaultText: true } }
        }
      })
    : [];

  const entries: Record<string, string> = { ...builtInEntries(activeLocale.code) };
  for (const value of values) {
    if (value.publishedText) {
      entries[`${value.key.namespace}.${value.key.key}`] = value.publishedText;
    }
  }

  return {
    locale: activeLocale.code,
    entries,
    t(key: string) {
      return entries[key] ?? key;
    }
  };
}
```

```ts
"use server";

import { revalidatePath } from "next/cache";
import { requireCurrentUserPermission } from "@/lib/current-user";
import { prisma } from "@/lib/db";
import { normalizeLocaleCode } from "@/lib/i18n/locale-codes";

function requiredString(formData: FormData, key: string) {
  const value = formData.get(key);
  if (typeof value !== "string" || value.trim().length === 0) {
    throw new Error(`Не заполнено поле ${key}.`);
  }
  return value.trim();
}

export async function createLocaleAction(formData: FormData) {
  const user = await requireCurrentUserPermission("admin:manage");
  const code = normalizeLocaleCode(requiredString(formData, "code"));
  const name = requiredString(formData, "name");

  await prisma.locale.create({
    data: {
      workspaceId: user.workspaceId,
      code,
      name,
      isEnabled: true,
      isDefault: false
    }
  });

  revalidatePath("/admin/localization");
}

export async function saveTranslationDraftAction(formData: FormData) {
  const user = await requireCurrentUserPermission("admin:manage");
  const localeId = requiredString(formData, "localeId");
  const keyId = requiredString(formData, "keyId");
  const draftText = requiredString(formData, "draftText");

  await prisma.$transaction(async (tx) => {
    const existing = await tx.translationValue.findUnique({
      where: { workspaceId_localeId_keyId: { workspaceId: user.workspaceId, localeId, keyId } }
    });
    await tx.translationValue.upsert({
      where: { workspaceId_localeId_keyId: { workspaceId: user.workspaceId, localeId, keyId } },
      create: { workspaceId: user.workspaceId, localeId, keyId, draftText },
      update: { draftText }
    });
    await tx.translationAudit.create({
      data: {
        workspaceId: user.workspaceId,
        localeId,
        keyId,
        actorId: user.id,
        action: "draft_save",
        beforeText: existing?.draftText ?? null,
        afterText: draftText
      }
    });
  });

  revalidatePath("/admin/localization");
}

export async function publishTranslationAction(formData: FormData) {
  const user = await requireCurrentUserPermission("admin:manage");
  const valueId = requiredString(formData, "valueId");

  await prisma.$transaction(async (tx) => {
    const value = await tx.translationValue.findUnique({
      where: { id: valueId },
      include: { locale: true, key: true }
    });
    if (!value || value.workspaceId !== user.workspaceId) {
      throw new Error("Перевод не найден.");
    }
    if (!value.draftText) {
      throw new Error("Черновик пуст.");
    }
    await tx.translationValue.update({
      where: { id: valueId },
      data: {
        publishedText: value.draftText,
        publishedAt: new Date(),
        publishedById: user.id,
        version: { increment: 1 }
      }
    });
    await tx.translationAudit.create({
      data: {
        workspaceId: user.workspaceId,
        localeId: value.localeId,
        keyId: value.keyId,
        actorId: user.id,
        action: "publish",
        beforeText: value.publishedText,
        afterText: value.draftText
      }
    });
  });

  revalidatePath("/admin/localization");
}

export async function rollbackTranslationAction(formData: FormData) {
  const user = await requireCurrentUserPermission("admin:manage");
  const valueId = requiredString(formData, "valueId");

  await prisma.$transaction(async (tx) => {
    const value = await tx.translationValue.findUnique({ where: { id: valueId } });
    if (!value || value.workspaceId !== user.workspaceId) {
      throw new Error("Перевод не найден.");
    }
    const previous = await tx.translationAudit.findFirst({
      where: {
        workspaceId: user.workspaceId,
        localeId: value.localeId,
        keyId: value.keyId,
        action: "publish",
        beforeText: { not: null }
      },
      orderBy: { createdAt: "desc" }
    });
    const restoredText = previous?.beforeText ?? null;
    await tx.translationValue.update({
      where: { id: valueId },
      data: {
        draftText: restoredText,
        publishedText: restoredText,
        publishedAt: restoredText ? new Date() : null,
        publishedById: restoredText ? user.id : null,
        version: { increment: 1 }
      }
    });
    await tx.translationAudit.create({
      data: {
        workspaceId: user.workspaceId,
        localeId: value.localeId,
        keyId: value.keyId,
        actorId: user.id,
        action: "rollback",
        beforeText: value.publishedText,
        afterText: restoredText
      }
    });
  });

  revalidatePath("/admin/localization");
}
```

```tsx
  publish(formData: FormData): Promise<void>;
  rollback(formData: FormData): Promise<void>;

  keys: KeyRow[];
  actions: Actions;

  const [filter, setFilter] = useState("");
  const rows = useMemo(() => {

    return normalized ? props.keys.filter((row) => row.fullKey.toLowerCase().includes(normalized)) : props.keys;
```

```ts
export type StatusTone = "positive" | "warning" | "negative" | "neutral" | "info";

export function toneForScore(score: number | null | undefined): StatusTone {
  if (score == null) return "neutral";
  if (score >= 90) return "positive";
  if (score >= 70) return "warning";
  return "negative";
}

export function toneForCount(
  count: number | null | undefined,
  tones: { zero: StatusTone; nonZero: StatusTone }
): StatusTone {
  return (count ?? 0) === 0 ? tones.zero : tones.nonZero;
}

export function statusToneClass(tone: StatusTone) {
  return `status-tone status-tone--${tone}`;
}
```
